			function blinkcaret() {
				var caret=document.getElementById("cp");
				
				if (caret==null) {
					return false;
				} else {
					var caretColor = caret.style.backgroundColor;
					caret.style.backgroundColor = (caretColor.replace(new RegExp("' '/g"), "") == "rgb(192,192,192)") ? "rgb(0, 0, 0)" : "rgb(192, 192, 192)";
				}
			}
						
			function setblinking(event) {
				try {
					setInterval(blinkcaret, 500);
				} catch (err) {
					return false;
				}
			}
