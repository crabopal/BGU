		<SCRIPT language="JScript" type="text/javascript">
			
			var localDragging = false;
			var copiedFragment = "";
			
			function dropping() {
				
				if (localDragging) {
					return;
				};
				
				textRange = document.selection.createRange();
				textToDrop = window.event.dataTransfer.getData("Text");
				
				if (textToDrop != "ПолеНастройки") {
					textRange.text = textToDrop;
				};
				
				textRange.select();
				window.event.returnValue = false;
			}
			
			function pasting() {
				
				if (window.clipboardData.getData("Text") == copiedFragment) {
					return true;
				} else {
					window.clipboardData.clearData("HTML");
					return true;
				}
			}
			
		</SCRIPT>
		
	</HEAD>
	
	<BODY ondragstart="localDragging=true"
		ondragend="localDragging=false"
		onpaste="pasting()"
		ondrop="dropping()"
		oncopy='copiedFragment=window.clipboardData.getData("Text")'
		oncut='copiedFragment=window.clipboardData.getData("Text")'
		onresizestart="return false"
